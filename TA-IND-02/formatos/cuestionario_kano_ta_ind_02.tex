\noindent Marque una respuesta en cada pregunta.

\begin{enumerate}[leftmargin=*, label=\textbf{\arabic*}.]

  \item \textbf{Registro de mascotas}\\
  Si el sistema permite registrar los datos de cada mascota, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}\\
  Si el sistema NO permite registrar los datos de cada mascota, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}

  \item \textbf{Historial de consultas}\\
  Si puede consultar todas las consultas anteriores de una mascota, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}\\
  Si NO puede consultar las consultas anteriores, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}

  \item \textbf{Registro de vacunas}\\
  Si el sistema permite guardar las vacunas aplicadas a cada mascota, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}\\
  Si NO permite guardar las vacunas aplicadas, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}

  \item \textbf{Avisos de próximas vacunas}\\
  Si el sistema le avisa cuándo una mascota debe recibir otra vacuna, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}\\
  Si el sistema NO envía esos avisos, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}

  \item \textbf{Búsqueda de historias clínicas}\\
  Si puede encontrar una historia clínica rápidamente mediante el nombre de la mascota, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}\\
  Si NO cuenta con esa búsqueda rápida, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}

  \item \textbf{Control de vacunas pendientes}\\
  Si el sistema muestra una lista de mascotas con vacunas pendientes, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}\\
  Si NO muestra esa lista, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}

  \item \textbf{Acceso restringido}\\
  Si solamente el personal autorizado puede modificar las historias clínicas, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}\\
  Si cualquier usuario del sistema pudiera modificarlas, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}

  \item \textbf{Descarga del historial}\\
  Si puede descargar o imprimir el historial clínico de una mascota, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}\\
  Si NO puede descargar ni imprimir ese historial, ¿cómo se sentiría? \rule{1.2cm}{0.4pt}

\end{enumerate}
